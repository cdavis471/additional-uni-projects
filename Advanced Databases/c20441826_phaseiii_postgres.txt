-- RANGE PARTITIONING (by date_key) | Partitions By Week
-- -----------------------------------------------------

-- Create Table
-- ------------
DROP TABLE IF EXISTS c20441826_phaseii_dw.fact_pricing_range CASCADE;
CREATE TABLE c20441826_phaseii_dw.fact_pricing_range ( LIKE c20441826_phaseii_dw.fact_pricing ) PARTITION BY RANGE (date_key);

-- Create Weekly RANGE Partitions
-- ------------------------------
DO $$
DECLARE
	d date;
	d_start date;
	d_end date;
BEGIN
	SELECT date_trunc('week', MIN(date_actual))::date, date_trunc('week', MAX(date_actual))::date
	INTO d_start, d_end
	FROM c20441826_phaseii_dw.dim_date;
	FOR d IN
		SELECT generate_series(d_start, d_end, interval '1 week')::date
	LOOP
		EXECUTE format(
			'CREATE TABLE c20441826_phaseii_dw.fact_pricing_range_w%s PARTITION OF c20441826_phaseii_dw.fact_pricing_range FOR VALUES FROM (%s) TO (%s);',
			to_char(d,'YYYYMMDD'),
			to_char(d,'YYYYMMDD')::int,
			to_char(d + 7,'YYYYMMDD')::int
		);
	END LOOP;
END$$;

-- RANGE TESTING
-- 1: Pre-Partition Test
-- 2: Post-Partition Test
-- ----------------------

-- Part 1
EXPLAIN (ANALYZE, BUFFERS)
WITH bounds AS (
  SELECT
    to_char(date_trunc('week', MIN(date_actual)),'YYYYMMDD')::int AS k_from,
    to_char(date_trunc('week', MIN(date_actual)) + interval '7 days','YYYYMMDD')::int AS k_to
  FROM c20441826_phaseii_dw.dim_date
)
SELECT COALESCE(SUM(fp.line_revenue),0) AS total_first_week
FROM c20441826_phaseii_dw.fact_pricing fp
CROSS JOIN bounds b
WHERE fp.date_key >= b.k_from AND fp.date_key < b.k_to;
